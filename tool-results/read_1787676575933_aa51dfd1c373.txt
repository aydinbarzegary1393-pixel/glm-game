   178→// ===== AUDIO SYSTEM (Web Audio API) =====
   179→let audioCtx = null;

   182→function ensureAudio() {
   183→  if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
   184→}
   185→
   186→function playFootstep() {
   187→  ensureAudio();
   188→  const vol = masterVolume * 0.3;
   189→  if (vol <= 0) return;
   190→  // Create short noise burst for footstep
   191→  const bufferSize = audioCtx.sampleRate * 0.08;
   192→  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
   193→  const data = buffer.getChannelData(0);
   194→  for (let i = 0; i < bufferSize; i++) {
   195→    data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.15));
   196→  }
   197→  const src = audioCtx.createBufferSource();
   198→  src.buffer = buffer;
   199→  const filter = audioCtx.createBiquadFilter();
   200→  filter.type = 'lowpass';
   201→  filter.frequency.value = 800;
   202→  const gain = audioCtx.createGain();
   203→  gain.gain.setValueAtTime(vol, audioCtx.currentTime);
   204→  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.08);
   205→  src.connect(filter).connect(gain).connect(audioCtx.destination);
   206→  src.start();
   207→}
   208→
   209→function playGunshot() {
   210→  ensureAudio();
   211→  const vol = masterVolume * 0.6;
   212→  if (vol <= 0) return;
   213→  const dur = 0.15;
   214→  const bufferSize = audioCtx.sampleRate * dur;
   215→  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
   216→  const data = buffer.getChannelData(0);
   217→  for (let i = 0; i < bufferSize; i++) {
   218→    const t = i / audioCtx.sampleRate;
   219→    data[i] = (Math.random() * 2 - 1) * Math.exp(-t * 30) * 1.5;
   220→  }
   221→  const src = audioCtx.createBufferSource();
   222→  src.buffer = buffer;
   223→  // Low thump
   224→  const osc = audioCtx.createOscillator();
   225→  osc.type = 'sine';
   226→  osc.frequency.setValueAtTime(150, audioCtx.currentTime);
   227→  osc.frequency.exponentialRampToValueAtTime(40, audioCtx.currentTime + 0.1);
   228→  const oscGain = audioCtx.createGain();
   229→  oscGain.gain.setValueAtTime(vol * 0.8, audioCtx.currentTime);
   230→  oscGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
   231→  // Noise part
   232→  const noiseGain = audioCtx.createGain();
   233→  noiseGain.gain.setValueAtTime(vol, audioCtx.currentTime);
   234→  noiseGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
   235→  const filter = audioCtx.createBiquadFilter();
   236→  filter.type = 'bandpass';
   237→  filter.frequency.value = 1000;
   238→  filter.Q.value = 0.5;
   239→  src.connect(filter).connect(noiseGain).connect(audioCtx.destination);
   240→  osc.connect(oscGain).connect(audioCtx.destination);
   241→  src.start();
   242→  osc.start();
   243→  osc.stop(audioCtx.currentTime + 0.1);
   244→}

   690→function shoot() {
   691→  if (!canShoot || ammo <= 0) return;
   692→  canShoot = false;
   693→  ammo--;
   694→
   695→  // Sound
   696→  playGunshot();
   697→
   698→  // Muzzle flash
   699→  muzzleFlashEl.classList.remove('fire');
   700→  void muzzleFlashEl.offsetWidth;
   701→  muzzleFlashEl.classList.add('fire');
   702→  muzzleLight.intensity = 3;
   703→  setTimeout(() => { muzzleLight.intensity = 0; }, 50);
   704→
   705→  // Gun recoil (Three.js, not DOM)
   706→  recoilAmount = 0.06;
   707→
   708→  // Raycast
   709→  raycaster.setFromCamera(new THREE.Vector2(0, 0), camera);
   710→  const hits = raycaster.intersectObjects(sceneObjects);
   711→  if (hits.length > 0) {
   712→    const hit = hits[0];
   713→    // Bullet hole effect
   714→    const holeGeo = new THREE.CircleGeometry(0.08, 8);
   715→    const holeMat = new THREE.MeshBasicMaterial({color:0x111111, side:THREE.DoubleSide});
   716→    const hole = new THREE.Mesh(holeGeo, holeMat);
   717→    hole.position.copy(hit.point);
   718→    hole.lookAt(hit.point.clone().add(hit.face.normal));
   719→    hole.position.add(hit.face.normal.clone().multiplyScalar(0.01));
   720→    scene.add(hole);
   721→    sceneObjects.push(hole);
   722→  }
   723→
   724→  // Fire rate cooldown
   725→  setTimeout(() => { canShoot = true; }, 120);
   726→}

   800→// ===== GAME LOOP =====
   801→const clock = new THREE.Clock();
   802→const debugEl = document.getElementById('debug');
   803→
   804→function animate() {
   805→  requestAnimationFrame(animate);
   806→  const dt = Math.min(clock.getDelta(), 0.05);
   807→
   808→  if (isLocked) {
   809→    // Movement
   810→    const fwd = new THREE.Vector3(0,0,-1).applyAxisAngle(new THREE.Vector3(0,1,0), yaw);
   811→    const rgt = new THREE.Vector3(1,0,0).applyAxisAngle(new THREE.Vector3(0,1,0), yaw);
   812→    const mv = new THREE.Vector3();
   813→    if(keys.w) mv.add(fwd); if(keys.s) mv.sub(fwd);
   814→    if(keys.d) mv.add(rgt); if(keys.a) mv.sub(rgt);
   815→    if(mv.lengthSq()>0) mv.normalize();
   816→
   817→    const moving = mv.lengthSq() > 0;
   818→    player.position.x += mv.x * MOVE_SPEED * dt;
   819→    player.position.z += mv.z * MOVE_SPEED * dt;
   820→
   821→    // Gravity
   822→    velocityY -= GRAVITY * dt;
   823→    player.position.y += velocityY * dt;
   824→    if(player.position.y <= PLAYER_HEIGHT) { player.position.y = PLAYER_HEIGHT; velocityY = 0; }
   825→
   826→    // Footstep sounds
   827→    if (moving) {
   828→      footstepTimer -= dt;
   829→      if (footstepTimer <= 0) {
   830→        playFootstep();
   831→        footstepTimer = FOOTSTEP_INTERVAL;
   832→      }
   833→    } else {
   834→      footstepTimer = 0;
   835→    }
   836→
   837→    // Gun recoil
   838→    if (recoilAmount > 0) {
   839→      recoilAmount *= 0.85;
   840→      if (recoilAmount < 0.001) recoilAmount = 0;
   841→    }
   842→
   843→    // Gun bob when walking
   844→    let bobY = 0, bobX = 0;
   845→    if (moving) {
   846→      const bobSpeed = 10;
   847→      const bobAmount = 0.008;
   848→      bobY = Math.sin(clock.elapsedTime * bobSpeed) * bobAmount;
   849→      bobX = Math.cos(clock.elapsedTime * bobSpeed * 0.5) * bobAmount * 0.5;
   850→    }
   851→    gunGroup.position.y = -0.22 + bobY + recoilAmount;
   852→    gunGroup.position.x = 0.25 + bobX;
   853→
   854→    // Debug + ammo HUD
   855→    debugEl.innerHTML =
   856→      'Pos: '+player.position.x.toFixed(1)+', '+player.position.y.toFixed(1)+', '+player.position.z.toFixed(1)+
   857→      '<br>Yaw: '+(yaw*180/Math.PI).toFixed(1)+'°  Pitch: '+(pitch*180/Math.PI).toFixed(1)+'°'+
   858→      '<br>Ammo: ' + ammo + ' / ' + maxAmmo;
   859→  }
   860→
   861→  renderer.render(scene, camera);
   862→}
   863→animate();

   264→// ===== AUDIO SYSTEM (Web Audio API) =====
   265→let audioCtx = null;
   266→let gunAudioCtx = null;
   267→
   268→function ensureAudio() {
   269→  if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
   270→}
   271→
   272→function playFootstep() {
   273→  ensureAudio();
   274→  const vol = masterVolume * 0.3;
   275→  if (vol <= 0) return;
   276→  // Create short noise burst for footstep
   277→  const bufferSize = audioCtx.sampleRate * 0.08;
   278→  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
   279→  const data = buffer.getChannelData(0);
   280→  for (let i = 0; i < bufferSize; i++) {
   281→    data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.15));
   282→  }
   283→  const src = audioCtx.createBufferSource();
   284→  src.buffer = buffer;
   285→  const filter = audioCtx.createBiquadFilter();
   286→  filter.type = 'lowpass';
   287→  filter.frequency.value = 800;
   288→  const gain = audioCtx.createGain();
   289→  gain.gain.setValueAtTime(vol, audioCtx.currentTime);
   290→  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.08);
   291→  src.connect(filter).connect(gain).connect(audioCtx.destination);
   292→  src.start();
   293→}
   294→
   295→function playGunshot() {
   296→  ensureAudio();
   297→  const vol = masterVolume * 0.6;
   298→  if (vol <= 0) return;
   299→  const dur = 0.15;
   300→  const bufferSize = audioCtx.sampleRate * dur;
   301→  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
   302→  const data = buffer.getChannelData(0);
   303→  for (let i = 0; i < bufferSize; i++) {
   304→    const t = i / audioCtx.sampleRate;
   305→    data[i] = (Math.random() * 2 - 1) * Math.exp(-t * 30) * 1.5;
   306→  }
   307→  const src = audioCtx.createBufferSource();
   308→  src.buffer = buffer;
   309→  // Low thump
   310→  const osc = audioCtx.createOscillator();
   311→  osc.type = 'sine';
   312→  osc.frequency.setValueAtTime(150, audioCtx.currentTime);
   313→  osc.frequency.exponentialRampToValueAtTime(40, audioCtx.currentTime + 0.1);
   314→  const oscGain = audioCtx.createGain();
   315→  oscGain.gain.setValueAtTime(vol * 0.8, audioCtx.currentTime);
   316→  oscGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
   317→  // Noise part
   318→  const noiseGain = audioCtx.createGain();
   319→  noiseGain.gain.setValueAtTime(vol, audioCtx.currentTime);
   320→  noiseGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
   321→  const filter = audioCtx.createBiquadFilter();
   322→  filter.type = 'bandpass';
   323→  filter.frequency.value = 1000;
   324→  filter.Q.value = 0.5;
   325→  src.connect(filter).connect(noiseGain).connect(audioCtx.destination);
   326→  osc.connect(oscGain).connect(audioCtx.destination);
   327→  src.start();
   328→  osc.start();
   329→  osc.stop(audioCtx.currentTime + 0.1);
   330→}

   332→function playReloadSound() {
   333→  ensureAudio();
   334→  const vol = masterVolume * 0.4;
   335→  if (vol <= 0) return;
   336→  // Click sound
   337→  var buf = audioCtx.createBuffer(1, audioCtx.sampleRate * 0.05, audioCtx.sampleRate);
   338→  var d = buf.getChannelData(0);
   339→  for (var i = 0; i < d.length; i++) d[i] = (Math.random() * 2 - 1) * Math.exp(-i / (d.length * 0.1));
   340→  var s = audioCtx.createBufferSource(); s.buffer = buf;
   341→  var g = audioCtx.createGain(); g.gain.setValueAtTime(vol, audioCtx.currentTime);
   342→  var f = audioCtx.createBiquadFilter(); f.type = 'bandpass'; f.frequency.value = 2000; f.Q.value = 2;
   343→  s.connect(f).connect(g).connect(audioCtx.destination); s.start();
   344→  // Magazine slide sound after delay
   345→  setTimeout(function() {
   346→    var buf2 = audioCtx.createBuffer(1, audioCtx.sampleRate * 0.15, audioCtx.sampleRate);
   347→    var d2 = buf2.getChannelData(0);
   348→    for (var i = 0; i < d2.length; i++) d2[i] = (Math.random() * 2 - 1) * Math.exp(-i / (d2.length * 0.15)) * 0.6;
   349→    var s2 = audioCtx.createBufferSource(); s2.buffer = buf2;
   350→    var g2 = audioCtx.createGain(); g2.gain.setValueAtTime(vol * 0.7, audioCtx.currentTime);
   351→    var f2 = audioCtx.createBiquadFilter(); f2.type = 'bandpass'; f2.frequency.value = 1500; f2.Q.value = 1;
   352→    s2.connect(f2).connect(g2).connect(audioCtx.destination); s2.start();
   353→  }, 300);
   354→}

   411→function initJoystick() {

   415→  zone.addEventListener('touchstart', function(e) {

   417→    var t = e.changedTouches[0];
   418→    joyTouchId = t.identifier;
   419→    var rect = joyBase.getBoundingClientRect();

   421→    joyBaseY = rect.top + rect.height / 2;

   424→    e.preventDefault();
   425→    for (var i = 0; i < e.changedTouches.length; i++) {
   426→      var t = e.changedTouches[i];
   427→      if (t.identifier === joyTouchId) {
   428→        var dx = t.clientX - joyBaseX;
   429→        var dy = t.clientY - joyBaseY;
   430→        var dist = Math.sqrt(dx * dx + dy * dy);
   431→        if (dist > JOY_RADIUS) { dx = dx / dist * JOY_RADIUS; dy = dy / dist * JOY_RADIUS; }
   432→        joyInputX = dx / JOY_RADIUS;
   433→        joyInputY = dy / JOY_RADIUS;
   434→        joyStick.style.transform = 'translate(' + dx + 'px,' + dy + 'px)';
   435→      }
   436→    }
   437→  }, {passive: false});

   439→    for (var i = 0; i < e.changedTouches.length; i++) {
   440→      if (e.changedTouches[i].identifier === joyTouchId) {
   441→        joyTouchId = null; joyInputX = 0; joyInputY = 0;
   442→        joyStick.style.transform = 'translate(0,0)';
   443→      }
   444→    }

   845→// ===== COLLISION DETECTION =====
   846→function checkCollision(px, py, pz) {
   847→  const r = PLAYER_RADIUS;
   848→  const feetY = py - PLAYER_HEIGHT;
   849→  for (let i = 0; i < colliders.length; i++) {
   850→    const c = colliders[i];
   851→    if (px + r > c.minX && px - r < c.maxX &&
   852→        pz + r > c.minZ && pz - r < c.maxZ) {
   853→      // Only block if collider extends at least 0.3 above player feet
   854→      // This prevents thin floor slabs from blocking movement
   855→      if (c.maxY > feetY + 0.3 && c.minY < py) {
   856→        return c;
   857→      }
   858→    }
   859→  }
   860→  return null;
   861→}

   863→// Find highest surface the player can stand on
   864→function getGroundY(px, py, pz) {
   865→  const r = PLAYER_RADIUS;
   866→  const feetY = py - PLAYER_HEIGHT;
   867→  let best = -999;
   868→  const STEP_UP = 0.4;
   869→  for (let i = 0; i < colliders.length; i++) {
   870→    const c = colliders[i];
   871→    if (px + r > c.minX && px - r < c.maxX &&
   872→        pz + r > c.minZ && pz - r < c.maxZ) {
   873→      // Only count surfaces near the TOP of the collider (player stands on it)
   874→      // Not walls where feet are at the BOTTOM
   875→      if (feetY >= c.maxY - STEP_UP && feetY <= c.maxY + 0.1 &&
   876→          c.minY < py && c.maxY > best) {
   877→        best = c.maxY;
   878→      }
   879→    }
   880→  }
   881→  return best;
   882→}

  1322→function shoot() {
  1323→  if (!canShoot || isReloading || ammo <= 0) return;
  1324→  canShoot = false;
  1325→  ammo--;
  1326→
  1327→  // Sound
  1328→  playGunshot();
  1329→
  1330→  // Muzzle flash
  1331→  muzzleFlashEl.classList.remove('fire');
  1332→  void muzzleFlashEl.offsetWidth;
  1333→  muzzleFlashEl.classList.add('fire');
  1334→  muzzleLight.intensity = 3;
  1335→  setTimeout(() => { muzzleLight.intensity = 0; }, 50);
  1336→
  1337→  // Gun recoil (Three.js, not DOM)
  1338→  recoilAmount = 0.06;
  1339→
  1340→  // Raycast
  1341→  raycaster.setFromCamera(new THREE.Vector2(0, 0), camera);
  1342→  const hits = raycaster.intersectObjects(sceneObjects);
  1343→  if (hits.length > 0) {
  1344→    const hit = hits[0];
  1345→    // Convert face normal to world space (fixes rotated objects like ground plane)
  1346→    const n = hit.face.normal.clone().transformDirection(hit.object.matrixWorld).normalize();
  1347→    // Bullet hole: dark center + lighter ring
  1348→    const holeGroup = new THREE.Group();
  1349→    const coreGeo = new THREE.CircleGeometry(0.06, 8);
  1350→    const coreMat = new THREE.MeshBasicMaterial({color:0x111111, side:THREE.DoubleSide, depthWrite:false});
  1351→    const core = new THREE.Mesh(coreGeo, coreMat);
  1352→    holeGroup.add(core);
  1353→    const ringGeo = new THREE.RingGeometry(0.06, 0.1, 12);
  1354→    const ringMat = new THREE.MeshBasicMaterial({color:0x333333, side:THREE.DoubleSide, depthWrite:false, transparent:true, opacity:0.6});
  1355→    const ring = new THREE.Mesh(ringGeo, ringMat);
  1356→    holeGroup.add(ring);
  1357→    holeGroup.position.copy(hit.point);
  1358→    holeGroup.lookAt(hit.point.clone().add(n));
  1359→    holeGroup.position.add(n.clone().multiplyScalar(0.02));
  1360→    holeGroup.renderOrder = 999;
  1361→    bulletHoles.add(holeGroup);
  1362→  }
  1363→
  1364→  // Fire rate cooldown
  1365→  setTimeout(() => { canShoot = true; }, 120);
  1366→}

  1512→      mv.add(fwd.clone().multiplyScalar(-joyInputY));

  1514→    }
  1515→    if(mv.lengthSq()>0) mv.normalize();
  1516→
  1517→    const speed = MOVE_SPEED * dt;
  1518→    const px = player.position.x;
  1519→    const py = player.position.y;
  1520→    const pz = player.position.z;
  1521→
  1522→    // Try X movement (with collision)
  1523→    let newX = px + mv.x * speed;
  1524→    if (!checkCollision(newX, py, pz)) {
  1525→      player.position.x = newX;
  1526→    }
  1527→
  1528→    // Try Z movement (with collision)
  1529→    let newZ = pz + mv.z * speed;
  1530→    if (!checkCollision(player.position.x, py, newZ)) {
  1531→      player.position.z = newZ;
  1532→    }
  1533→
  1534→    const moving = mv.lengthSq() > 0;
  1535→

  1538→    var groundLevel = rawGround > -998 ? rawGround : 0;
  1539→    var onGround = (player.position.y - PLAYER_HEIGHT <= groundLevel + 0.05 && velocityY <= 0.1);
  1540→    if (keys.space && onGround) {
  1541→      velocityY = JUMP_FORCE;
  1542→    }
  1543→    // Gravity
  1544→    velocityY -= GRAVITY * dt;
  1545→    player.position.y += velocityY * dt;
  1546→    // Ground collision: stand on boxes/floors
  1547→    rawGround = getGroundY(player.position.x, player.position.y, player.position.z);
  1548→    groundLevel = rawGround > -998 ? rawGround : 0;
  1549→    if (player.position.y - PLAYER_HEIGHT <= groundLevel + 0.05) {
  1550→      player.position.y = groundLevel + PLAYER_HEIGHT;
  1551→      velocityY = 0;

  1554→    // Footstep sounds

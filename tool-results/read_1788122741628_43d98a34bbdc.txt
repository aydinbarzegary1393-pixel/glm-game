   257→// ===== AUDIO SYSTEM (Web Audio API) =====
   258→let audioCtx = null;
   259→let gunAudioCtx = null;
   260→
   261→function ensureAudio() {
   262→  if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
   263→}
   264→
   265→function playFootstep() {
   266→  ensureAudio();
   267→  const vol = masterVolume * 0.3;
   268→  if (vol <= 0) return;
   269→  // Create short noise burst for footstep
   270→  const bufferSize = audioCtx.sampleRate * 0.08;
   271→  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
   272→  const data = buffer.getChannelData(0);
   273→  for (let i = 0; i < bufferSize; i++) {
   274→    data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.15));
   275→  }
   276→  const src = audioCtx.createBufferSource();
   277→  src.buffer = buffer;
   278→  const filter = audioCtx.createBiquadFilter();
   279→  filter.type = 'lowpass';
   280→  filter.frequency.value = 800;
   281→  const gain = audioCtx.createGain();
   282→  gain.gain.setValueAtTime(vol, audioCtx.currentTime);
   283→  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.08);
   284→  src.connect(filter).connect(gain).connect(audioCtx.destination);
   285→  src.start();
   286→}
   287→
   288→function playGunshot() {
   289→  ensureAudio();
   290→  const vol = masterVolume * 0.6;
   291→  if (vol <= 0) return;
   292→  const dur = 0.15;
   293→  const bufferSize = audioCtx.sampleRate * dur;
   294→  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
   295→  const data = buffer.getChannelData(0);
   296→  for (let i = 0; i < bufferSize; i++) {
   297→    const t = i / audioCtx.sampleRate;
   298→    data[i] = (Math.random() * 2 - 1) * Math.exp(-t * 30) * 1.5;
   299→  }
   300→  const src = audioCtx.createBufferSource();
   301→  src.buffer = buffer;
   302→  // Low thump
   303→  const osc = audioCtx.createOscillator();
   304→  osc.type = 'sine';
   305→  osc.frequency.setValueAtTime(150, audioCtx.currentTime);
   306→  osc.frequency.exponentialRampToValueAtTime(40, audioCtx.currentTime + 0.1);
   307→  const oscGain = audioCtx.createGain();
   308→  oscGain.gain.setValueAtTime(vol * 0.8, audioCtx.currentTime);
   309→  oscGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
   310→  // Noise part
   311→  const noiseGain = audioCtx.createGain();
   312→  noiseGain.gain.setValueAtTime(vol, audioCtx.currentTime);
   313→  noiseGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
   314→  const filter = audioCtx.createBiquadFilter();
   315→  filter.type = 'bandpass';
   316→  filter.frequency.value = 1000;

   318→  src.connect(filter).connect(noiseGain).connect(audioCtx.destination);
   319→  osc.connect(oscGain).connect(audioCtx.destination);

   325→function playReloadSound() {
   326→  ensureAudio();
   327→  const vol = masterVolume * 0.4;
   328→  if (vol <= 0) return;
   329→  // Click sound
   330→  var buf = audioCtx.createBuffer(1, audioCtx.sampleRate * 0.05, audioCtx.sampleRate);
   331→  var d = buf.getChannelData(0);
   332→  for (var i = 0; i < d.length; i++) d[i] = (Math.random() * 2 - 1) * Math.exp(-i / (d.length * 0.1));
   333→  var s = audioCtx.createBufferSource(); s.buffer = buf;
   334→  var g = audioCtx.createGain(); g.gain.setValueAtTime(vol, audioCtx.currentTime);
   335→  var f = audioCtx.createBiquadFilter(); f.type = 'bandpass'; f.frequency.value = 2000; f.Q.value = 2;
   336→  s.connect(f).connect(g).connect(audioCtx.destination); s.start();
   337→  // Magazine slide sound after delay
   338→  setTimeout(function() {
   339→    var buf2 = audioCtx.createBuffer(1, audioCtx.sampleRate * 0.15, audioCtx.sampleRate);
   340→    var d2 = buf2.getChannelData(0);
   341→    for (var i = 0; i < d2.length; i++) d2[i] = (Math.random() * 2 - 1) * Math.exp(-i / (d2.length * 0.15)) * 0.6;
   342→    var s2 = audioCtx.createBufferSource(); s2.buffer = buf2;
   343→    var g2 = audioCtx.createGain(); g2.gain.setValueAtTime(vol * 0.7, audioCtx.currentTime);
   344→    var f2 = audioCtx.createBiquadFilter(); f2.type = 'bandpass'; f2.frequency.value = 1500; f2.Q.value = 1;
   345→    s2.connect(f2).connect(g2).connect(audioCtx.destination); s2.start();
   346→  }, 300);
   347→}

   401→        if (dist > JOY_RADIUS) { dx = dx / dist * JOY_RADIUS; dy = dy / dist * JOY_RADIUS; }

   793→// ===== COLLISION DETECTION =====
   794→function checkCollision(px, py, pz) {
   795→  const r = PLAYER_RADIUS;
   796→  const feetY = py - PLAYER_HEIGHT;
   797→  for (let i = 0; i < colliders.length; i++) {
   798→    const c = colliders[i];
   799→    if (px + r > c.minX && px - r < c.maxX &&
   800→        pz + r > c.minZ && pz - r < c.maxZ) {
   801→      // Only block if collider extends at least 0.3 above player feet
   802→      // This prevents thin floor slabs from blocking movement
   803→      if (c.maxY > feetY + 0.3 && c.minY < py) {
   804→        return c;
   805→      }
   806→    }
   807→  }
   808→  return null;
   809→}
   810→
   811→// Find highest surface the player can stand on
   812→function getGroundY(px, py, pz) {
   813→  const r = PLAYER_RADIUS;
   814→  const feetY = py - PLAYER_HEIGHT;
   815→  let best = -999;
   816→  const STEP_UP = 0.4;
   817→  for (let i = 0; i < colliders.length; i++) {
   818→    const c = colliders[i];
   819→    if (px + r > c.minX && px - r < c.maxX &&
   820→        pz + r > c.minZ && pz - r < c.maxZ) {
   821→      // Only count surfaces near the TOP of the collider (player stands on it)
   822→      // Not walls where feet are at the BOTTOM
   823→      if (feetY >= c.maxY - STEP_UP && feetY <= c.maxY + 0.1 &&
   824→          c.minY < py && c.maxY > best) {
   825→        best = c.maxY;
   826→      }
   827→    }
   828→  }
   829→  return best;
   830→}

  1445→function animate() {
  1446→  requestAnimationFrame(animate);
  1447→  const dt = Math.min(clock.getDelta(), 0.05);
  1448→
  1449→  if (isLocked || (isMobileDevice && gameStarted)) {
  1450→    // Movement
  1451→    const fwd = new THREE.Vector3(0,0,-1).applyAxisAngle(new THREE.Vector3(0,1,0), yaw);
  1452→    const rgt = new THREE.Vector3(1,0,0).applyAxisAngle(new THREE.Vector3(0,1,0), yaw);
  1453→    const mv = new THREE.Vector3();
  1454→    if(keys.w) mv.add(fwd); if(keys.s) mv.sub(fwd);
  1455→    if(keys.d) mv.add(rgt); if(keys.a) mv.sub(rgt);
  1456→    // Mobile joystick input
  1457→    if (isMobileDevice && (Math.abs(joyInputX) > 0.1 || Math.abs(joyInputY) > 0.1)) {
  1458→      mv.add(fwd.clone().multiplyScalar(-joyInputY));
  1459→      mv.add(rgt.clone().multiplyScalar(joyInputX));
  1460→    }
  1461→    if(mv.lengthSq()>0) mv.normalize();
  1462→
  1463→    const speed = MOVE_SPEED * dt;
  1464→    const px = player.position.x;
  1465→    const py = player.position.y;
  1466→    const pz = player.position.z;
  1467→
  1468→    // Try X movement (with collision)
  1469→    let newX = px + mv.x * speed;
  1470→    if (!checkCollision(newX, py, pz)) {
  1471→      player.position.x = newX;
  1472→    }
  1473→
  1474→    // Try Z movement (with collision)
  1475→    let newZ = pz + mv.z * speed;
  1476→    if (!checkCollision(player.position.x, py, newZ)) {
  1477→      player.position.z = newZ;
  1478→    }
  1479→
  1480→    const moving = mv.lengthSq() > 0;
  1481→
  1482→    // Jump (only if standing on ground or a surface)
  1483→    var rawGround = getGroundY(player.position.x, player.position.y, player.position.z);
  1484→    var groundLevel = rawGround > -998 ? rawGround : 0;

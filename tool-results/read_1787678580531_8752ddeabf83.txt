   183→   183→function ensureAudio() {
   184→   184→  if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
   185→   185→}
   186→   186→
   187→   187→function playFootstep() {
   188→   188→  ensureAudio();
   189→   189→  const vol = masterVolume * 0.3;
   190→   190→  if (vol <= 0) return;
   191→   191→  // Create short noise burst for footstep
   192→   192→  const bufferSize = audioCtx.sampleRate * 0.08;
   193→   193→  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
   194→   194→  const data = buffer.getChannelData(0);
   195→   195→  for (let i = 0; i < bufferSize; i++) {
   196→   196→    data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.15));
   197→   197→  }
   198→   198→  const src = audioCtx.createBufferSource();
   199→   199→  src.buffer = buffer;
   200→   200→  const filter = audioCtx.createBiquadFilter();
   201→   201→  filter.type = 'lowpass';
   202→   202→  filter.frequency.value = 800;
   203→   203→  const gain = audioCtx.createGain();
   204→   204→  gain.gain.setValueAtTime(vol, audioCtx.currentTime);
   205→   205→  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.08);
   206→   206→  src.connect(filter).connect(gain).connect(audioCtx.destination);
   207→   207→  src.start();
   208→   208→}
   209→   209→
   210→   210→function playGunshot() {
   211→   211→  ensureAudio();
   212→   212→  const vol = masterVolume * 0.6;
   213→   213→  if (vol <= 0) return;
   214→   214→  const dur = 0.15;
   215→   215→  const bufferSize = audioCtx.sampleRate * dur;
   216→   216→  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
   217→   217→  const data = buffer.getChannelData(0);
   218→   218→  for (let i = 0; i < bufferSize; i++) {
   219→   219→    const t = i / audioCtx.sampleRate;
   220→   220→    data[i] = (Math.random() * 2 - 1) * Math.exp(-t * 30) * 1.5;
   221→   221→  }
   222→   222→  const src = audioCtx.createBufferSource();
   223→   223→  src.buffer = buffer;
   224→   224→  // Low thump
   225→   225→  const osc = audioCtx.createOscillator();
   226→   226→  osc.type = 'sine';
   227→   227→  osc.frequency.setValueAtTime(150, audioCtx.currentTime);
   228→   228→  osc.frequency.exponentialRampToValueAtTime(40, audioCtx.currentTime + 0.1);
   229→   229→  const oscGain = audioCtx.createGain();
   230→   230→  oscGain.gain.setValueAtTime(vol * 0.8, audioCtx.currentTime);
   231→   231→  oscGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
   232→   232→  // Noise part
   233→   233→  const noiseGain = audioCtx.createGain();
   234→   234→  noiseGain.gain.setValueAtTime(vol, audioCtx.currentTime);
   235→   235→  noiseGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
   236→   236→  const filter = audioCtx.createBiquadFilter();
   237→   237→  filter.type = 'bandpass';
   238→   238→  filter.frequency.value = 1000;
   239→   239→  filter.Q.value = 0.5;
   240→   240→  src.connect(filter).connect(noiseGain).connect(audioCtx.destination);
   241→   241→  osc.connect(oscGain).connect(audioCtx.destination);
   242→   242→  src.start();
   243→   243→  osc.start();
   244→   244→  osc.stop(audioCtx.currentTime + 0.1);
   245→   245→}

   386→   386→// ===== COLLISION DETECTION =====
   387→   387→function checkCollision(px, py, pz) {
   388→   388→  const r = PLAYER_RADIUS;
   389→   389→  for (let i = 0; i < colliders.length; i++) {
   390→   390→    const c = colliders[i];
   391→   391→    if (px + r > c.minX && px - r < c.maxX &&
   392→   392→        py - PLAYER_HEIGHT < c.maxY && py > c.minY &&
   393→   393→        pz + r > c.minZ && pz - r < c.maxZ) {
   394→   394→      return c;
   395→   395→    }
   396→   396→  }
   397→   397→  return null;
   398→   398→}

   935→   935→function animate() {
   936→   936→  requestAnimationFrame(animate);
   937→   937→  const dt = Math.min(clock.getDelta(), 0.05);
   938→   938→
   939→   939→  if (isLocked) {
   940→   940→    // Movement
   941→   941→    const fwd = new THREE.Vector3(0,0,-1).applyAxisAngle(new THREE.Vector3(0,1,0), yaw);
   942→   942→    const rgt = new THREE.Vector3(1,0,0).applyAxisAngle(new THREE.Vector3(0,1,0), yaw);
   943→   943→    const mv = new THREE.Vector3();
   944→   944→    if(keys.w) mv.add(fwd); if(keys.s) mv.sub(fwd);
   945→   945→    if(keys.d) mv.add(rgt); if(keys.a) mv.sub(rgt);
   946→   946→    if(mv.lengthSq()>0) mv.normalize();
   947→   947→
   948→   948→    const speed = MOVE_SPEED * dt;
   949→   949→    const px = player.position.x;
   950→   950→    const py = player.position.y;
   951→   951→    const pz = player.position.z;
   952→   952→
   953→   953→    // Try X movement (with collision)
   954→   954→    let newX = px + mv.x * speed;
   955→   955→    if (!checkCollision(newX, py, pz)) {
   956→   956→      player.position.x = newX;
   957→   957→    }
   958→   958→
   959→   959→    // Try Z movement (with collision)
   960→   960→    let newZ = pz + mv.z * speed;
   961→   961→    if (!checkCollision(player.position.x, py, newZ)) {
   962→   962→      player.position.z = newZ;
   963→   963→    }
   964→   964→
   965→   965→    const moving = mv.lengthSq() > 0;
   966→   966→
   967→   967→    // Gravity
   968→   968→    velocityY -= GRAVITY * dt;
   969→   969→    player.position.y += velocityY * dt;
   970→   970→    if(player.position.y <= PLAYER_HEIGHT) { player.position.y = PLAYER_HEIGHT; velocityY = 0; }
   971→   971→
   972→   972→    // Footstep sounds
   973→   973→    if (moving) {
   974→   974→      footstepTimer -= dt;
   975→   975→      if (footstepTimer <= 0) {
   976→   976→        playFootstep();
   977→   977→        footstepTimer = FOOTSTEP_INTERVAL;
   978→   978→      }
   979→   979→    } else {
   980→   980→      footstepTimer = 0;
   981→   981→    }
   982→   982→
   983→   983→    // Gun recoil
   984→   984→    if (recoilAmount > 0) {
   985→   985→      recoilAmount *= 0.85;
   986→   986→      if (recoilAmount < 0.001) recoilAmount = 0;
   987→   987→    }
   988→   988→
   989→   989→    // Gun bob when walking
   990→   990→    let bobY = 0, bobX = 0;
   991→   991→    if (moving) {
   992→   992→      const bobSpeed = 10;

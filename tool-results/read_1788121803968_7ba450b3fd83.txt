   257→// ===== AUDIO SYSTEM (Web Audio API) =====
   258→let audioCtx = null;
   259→let gunAudioCtx = null;
   260→
   261→function ensureAudio() {
   262→  if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
   263→}
   264→
   265→function playFootstep() {
   266→  ensureAudio();
   267→  const vol = masterVolume * 0.3;
   268→  if (vol <= 0) return;
   269→  // Create short noise burst for footstep
   270→  const bufferSize = audioCtx.sampleRate * 0.08;
   271→  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
   272→  const data = buffer.getChannelData(0);
   273→  for (let i = 0; i < bufferSize; i++) {
   274→    data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.15));
   275→  }
   276→  const src = audioCtx.createBufferSource();
   277→  src.buffer = buffer;
   278→  const filter = audioCtx.createBiquadFilter();
   279→  filter.type = 'lowpass';
   280→  filter.frequency.value = 800;
   281→  const gain = audioCtx.createGain();
   282→  gain.gain.setValueAtTime(vol, audioCtx.currentTime);
   283→  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.08);
   284→  src.connect(filter).connect(gain).connect(audioCtx.destination);
   285→  src.start();
   286→}
   287→
   288→function playGunshot() {
   289→  ensureAudio();
   290→  const vol = masterVolume * 0.6;
   291→  if (vol <= 0) return;
   292→  const dur = 0.15;
   293→  const bufferSize = audioCtx.sampleRate * dur;
   294→  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
   295→  const data = buffer.getChannelData(0);
   296→  for (let i = 0; i < bufferSize; i++) {
   297→    const t = i / audioCtx.sampleRate;
   298→    data[i] = (Math.random() * 2 - 1) * Math.exp(-t * 30) * 1.5;
   299→  }
   300→  const src = audioCtx.createBufferSource();
   301→  src.buffer = buffer;
   302→  // Low thump
   303→  const osc = audioCtx.createOscillator();
   304→  osc.type = 'sine';
   305→  osc.frequency.setValueAtTime(150, audioCtx.currentTime);
   306→  osc.frequency.exponentialRampToValueAtTime(40, audioCtx.currentTime + 0.1);
   307→  const oscGain = audioCtx.createGain();
   308→  oscGain.gain.setValueAtTime(vol * 0.8, audioCtx.currentTime);
   309→  oscGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
   310→  // Noise part
   311→  const noiseGain = audioCtx.createGain();
   312→  noiseGain.gain.setValueAtTime(vol, audioCtx.currentTime);
   313→  noiseGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
   314→  const filter = audioCtx.createBiquadFilter();
   315→  filter.type = 'bandpass';
   316→  filter.frequency.value = 1000;

   318→  src.connect(filter).connect(noiseGain).connect(audioCtx.destination);

   325→function playReloadSound() {
   326→  ensureAudio();
   327→  const vol = masterVolume * 0.4;
   328→  if (vol <= 0) return;
   329→  // Click sound
   330→  var buf = audioCtx.createBuffer(1, audioCtx.sampleRate * 0.05, audioCtx.sampleRate);
   331→  var d = buf.getChannelData(0);
   332→  for (var i = 0; i < d.length; i++) d[i] = (Math.random() * 2 - 1) * Math.exp(-i / (d.length * 0.1));
   333→  var s = audioCtx.createBufferSource(); s.buffer = buf;
   334→  var g = audioCtx.createGain(); g.gain.setValueAtTime(vol, audioCtx.currentTime);
   335→  var f = audioCtx.createBiquadFilter(); f.type = 'bandpass'; f.frequency.value = 2000; f.Q.value = 2;
   336→  s.connect(f).connect(g).connect(audioCtx.destination); s.start();
   337→  // Magazine slide sound after delay
   338→  setTimeout(function() {
   339→    var buf2 = audioCtx.createBuffer(1, audioCtx.sampleRate * 0.15, audioCtx.sampleRate);
   340→    var d2 = buf2.getChannelData(0);
   341→    for (var i = 0; i < d2.length; i++) d2[i] = (Math.random() * 2 - 1) * Math.exp(-i / (d2.length * 0.15)) * 0.6;
   342→    var s2 = audioCtx.createBufferSource(); s2.buffer = buf2;
   343→    var g2 = audioCtx.createGain(); g2.gain.setValueAtTime(vol * 0.7, audioCtx.currentTime);
   344→    var f2 = audioCtx.createBiquadFilter(); f2.type = 'bandpass'; f2.frequency.value = 1500; f2.Q.value = 1;
   345→    s2.connect(f2).connect(g2).connect(audioCtx.destination); s2.start();
   346→  }, 300);
   347→}

   396→      var t = e.changedTouches[i];
   397→      if (t.identifier === joyTouchId) {
   398→        var dx = t.clientX - joyBaseX;
   399→        var dy = t.clientY - joyBaseY;
   400→        var dist = Math.sqrt(dx * dx + dy * dy);
   401→        if (dist > JOY_RADIUS) { dx = dx / dist * JOY_RADIUS; dy = dy / dist * JOY_RADIUS; }
   402→        joyInputX = dx / JOY_RADIUS;
   403→        joyInputY = dy / JOY_RADIUS;

   412→        joyStick.style.transform = 'translate(0,0)';

   781→// ===== COLLISION DETECTION =====
   782→function checkCollision(px, py, pz) {
   783→  const r = PLAYER_RADIUS;
   784→  const feetY = py - PLAYER_HEIGHT;
   785→  for (let i = 0; i < colliders.length; i++) {
   786→    const c = colliders[i];
   787→    if (px + r > c.minX && px - r < c.maxX &&
   788→        pz + r > c.minZ && pz - r < c.maxZ) {
   789→      // Only block if collider extends at least 0.3 above player feet
   790→      // This prevents thin floor slabs from blocking movement
   791→      if (c.maxY > feetY + 0.3 && c.minY < py) {
   792→        return c;
   793→      }
   794→    }
   795→  }
   796→  return null;
   797→}

   799→// Find highest surface the player can stand on
   800→function getGroundY(px, py, pz) {
   801→  const r = PLAYER_RADIUS;
   802→  const feetY = py - PLAYER_HEIGHT;
   803→  let best = -999;
   804→  const STEP_UP = 0.4;
   805→  for (let i = 0; i < colliders.length; i++) {
   806→    const c = colliders[i];
   807→    if (px + r > c.minX && px - r < c.maxX &&
   808→        pz + r > c.minZ && pz - r < c.maxZ) {
   809→      // Only count surfaces near the TOP of the collider (player stands on it)
   810→      // Not walls where feet are at the BOTTOM
   811→      if (feetY >= c.maxY - STEP_UP && feetY <= c.maxY + 0.1 &&
   812→          c.minY < py && c.maxY > best) {
   813→        best = c.maxY;
   814→      }
   815→    }
   816→  }
   817→  return best;
   818→}

  1270→    hole.position.add(hit.face.normal.clone().multiplyScalar(0.01));

  1408→function animate() {
  1409→  requestAnimationFrame(animate);
  1410→  const dt = Math.min(clock.getDelta(), 0.05);
  1411→
  1412→  if (isLocked || (isMobileDevice && gameStarted)) {
  1413→    // Movement
  1414→    const fwd = new THREE.Vector3(0,0,-1).applyAxisAngle(new THREE.Vector3(0,1,0), yaw);
  1415→    const rgt = new THREE.Vector3(1,0,0).applyAxisAngle(new THREE.Vector3(0,1,0), yaw);
  1416→    const mv = new THREE.Vector3();
  1417→    if(keys.w) mv.add(fwd); if(keys.s) mv.sub(fwd);
  1418→    if(keys.d) mv.add(rgt); if(keys.a) mv.sub(rgt);
  1419→    // Mobile joystick input
  1420→    if (isMobileDevice && (Math.abs(joyInputX) > 0.1 || Math.abs(joyInputY) > 0.1)) {
  1421→      mv.add(fwd.clone().multiplyScalar(-joyInputY));
  1422→      mv.add(rgt.clone().multiplyScalar(joyInputX));
  1423→    }
  1424→    if(mv.lengthSq()>0) mv.normalize();
  1425→
  1426→    const speed = MOVE_SPEED * dt;
  1427→    const px = player.position.x;
  1428→    const py = player.position.y;
  1429→    const pz = player.position.z;
  1430→
  1431→    // Try X movement (with collision)
  1432→    let newX = px + mv.x * speed;
  1433→    if (!checkCollision(newX, py, pz)) {
  1434→      player.position.x = newX;
  1435→    }
  1436→
  1437→    // Try Z movement (with collision)
  1438→    let newZ = pz + mv.z * speed;
  1439→    if (!checkCollision(player.position.x, py, newZ)) {
  1440→      player.position.z = newZ;
  1441→    }
  1442→
  1443→    const moving = mv.lengthSq() > 0;
  1444→
  1445→    // Jump (only if standing on ground or a surface)
  1446→    var rawGround = getGroundY(player.position.x, player.position.y, player.position.z);
  1447→    var groundLevel = rawGround > -998 ? rawGround : 0;
  1448→    var onGround = (player.position.y - PLAYER_HEIGHT <= groundLevel + 0.05 && velocityY <= 0.1);
  1449→    if (keys.space && onGround) {
  1450→      velocityY = JUMP_FORCE;
  1451→    }
  1452→    // Gravity
  1453→    velocityY -= GRAVITY * dt;
  1454→    player.position.y += velocityY * dt;
  1455→    // Ground collision: stand on boxes/floors
  1456→    rawGround = getGroundY(player.position.x, player.position.y, player.position.z);
  1457→    groundLevel = rawGround > -998 ? rawGround : 0;
  1458→    if (player.position.y - PLAYER_HEIGHT <= groundLevel + 0.05) {
  1459→      player.position.y = groundLevel + PLAYER_HEIGHT;
  1460→      velocityY = 0;
  1461→    }
  1462→
  1463→    // Footstep sounds
  1464→    if (moving) {
  1465→      footstepTimer -= dt;
  1466→      if (footstepTimer <= 0) {
  1467→        playFootstep();
  1468→        footstepTimer = FOOTSTEP_INTERVAL;
  1469→      }
  1470→    } else {
  1471→      footstepTimer = 0;
  1472→    }
  1473→
  1474→    // Gun recoil
  1475→    if (recoilAmount > 0) {
  1476→      recoilAmount *= 0.85;
  1477→      if (recoilAmount < 0.001) recoilAmount = 0;
  1478→    }
  1479→
  1480→    // Gun bob when walking
  1481→    let bobY = 0, bobX = 0;
  1482→    if (moving) {
  1483→      const bobSpeed = 10;
  1484→      const bobAmount = 0.008;
  1485→      bobY = Math.sin(clock.elapsedTime * bobSpeed) * bobAmount;
  1486→      bobX = Math.cos(clock.elapsedTime * bobSpeed * 0.5) * bobAmount * 0.5;
  1487→    }
  1488→    gunGroup.position.y = -0.22 + bobY + recoilAmount;
  1489→    gunGroup.position.x = 0.25 + bobX;
  1490→
  1491→    // Debug + ammo HUD
  1492→    debugEl.innerHTML =
  1493→      'Pos: '+player.position.x.toFixed(1)+', '+player.position.y.toFixed(1)+', '+player.position.z.toFixed(1)+
  1494→      '<br>Yaw: '+(yaw*180/Math.PI).toFixed(1)+'°  Pitch: '+(pitch*180/Math.PI).toFixed(1)+'°'+

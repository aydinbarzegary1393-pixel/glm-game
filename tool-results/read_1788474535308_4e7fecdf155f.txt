   351→function playFootstep() {
   352→  ensureAudio();
   353→  const vol = masterVolume * 0.3;
   354→  if (vol <= 0) return;
   355→  // Create short noise burst for footstep
   356→  const bufferSize = audioCtx.sampleRate * 0.08;
   357→  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
   358→  const data = buffer.getChannelData(0);
   359→  for (let i = 0; i < bufferSize; i++) {
   360→    data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.15));
   361→  }
   362→  const src = audioCtx.createBufferSource();
   363→  src.buffer = buffer;
   364→  const filter = audioCtx.createBiquadFilter();
   365→  filter.type = 'lowpass';
   366→  filter.frequency.value = 800;
   367→  const gain = audioCtx.createGain();
   368→  gain.gain.setValueAtTime(vol, audioCtx.currentTime);
   369→  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.08);
   370→  src.connect(filter).connect(gain).connect(audioCtx.destination);
   371→  src.start();
   372→}

   374→function playGunshot() {
   375→  ensureAudio();
   376→  const vol = masterVolume * 0.6;
   377→  if (vol <= 0) return;
   378→  const dur = 0.15;
   379→  const bufferSize = audioCtx.sampleRate * dur;
   380→  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
   381→  const data = buffer.getChannelData(0);
   382→  for (let i = 0; i < bufferSize; i++) {
   383→    const t = i / audioCtx.sampleRate;
   384→    data[i] = (Math.random() * 2 - 1) * Math.exp(-t * 30) * 1.5;
   385→  }
   386→  const src = audioCtx.createBufferSource();
   387→  src.buffer = buffer;
   388→  // Low thump
   389→  const osc = audioCtx.createOscillator();
   390→  osc.type = 'sine';
   391→  osc.frequency.setValueAtTime(150, audioCtx.currentTime);
   392→  osc.frequency.exponentialRampToValueAtTime(40, audioCtx.currentTime + 0.1);
   393→  const oscGain = audioCtx.createGain();
   394→  oscGain.gain.setValueAtTime(vol * 0.8, audioCtx.currentTime);
   395→  oscGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
   396→  // Noise part
   397→  const noiseGain = audioCtx.createGain();
   398→  noiseGain.gain.setValueAtTime(vol, audioCtx.currentTime);
   399→  noiseGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
   400→  const filter = audioCtx.createBiquadFilter();
   401→  filter.type = 'bandpass';
   402→  filter.frequency.value = 1000;
   403→  filter.Q.value = 0.5;
   404→  src.connect(filter).connect(noiseGain).connect(audioCtx.destination);
   405→  osc.connect(oscGain).connect(audioCtx.destination);
   406→  src.start();
   407→  osc.start();
   408→  osc.stop(audioCtx.currentTime + 0.1);
   409→}

   506→    e.preventDefault();

   508→      var t = e.changedTouches[i];
   509→      if (t.identifier === joyTouchId) {
   510→        var dx = t.clientX - joyBaseX;
   511→        var dy = t.clientY - joyBaseY;
   512→        var dist = Math.sqrt(dx * dx + dy * dy);
   513→        if (dist > JOY_RADIUS) { dx = dx / dist * JOY_RADIUS; dy = dy / dist * JOY_RADIUS; }

   691→function checkEnemyWall(px, pz) {
   692→  var r = 0.5;
   693→  for (var i = 0; i < colliders.length; i++) {
   694→    var c = colliders[i];
   695→    if (c.maxY - c.minY < 0.5) continue;
   696→    if (c.maxY < 0.1 || c.minY > 2.5) continue;
   697→    if (px + r > c.minX && px - r < c.maxX &&
   698→        pz + r > c.minZ && pz - r < c.maxZ) return true;
   699→  }
   700→  return false;
   701→}

   774→      var zone = zones[Math.floor(Math.random() * zones.length)];

   776→      ez = zone.z1 + Math.random() * (zone.z2 - zone.z1);
   777→      if (!checkEnemyWall(ex, ez)) {

   781→      attempts++;
   782→    } while (!valid && attempts < 50);
   783→    if (valid) createEnemy(ex, ez);
   784→  }

   842→function updateEnemies(dt) {
   843→  var pPos = player.position;
   844→  for (var i = enemies.length - 1; i >= 0; i--) {
   845→    var e = enemies[i];
   846→    var dx = pPos.x - e.group.position.x;
   847→    var dz = pPos.z - e.group.position.z;
   848→    var dist = Math.sqrt(dx*dx + dz*dz);
   849→    if (dist < 0.1) dist = 0.1;
   850→    e.group.rotation.y = Math.atan2(dx, dz);
   851→    if (dist > ENEMY_ATK_RANGE) {
   852→      var speed = ENEMY_SPEED * dt;
   853→      var nx = dx/dist, nz = dz/dist;
   854→      var newX = e.group.position.x + nx*speed;
   855→      var newZ = e.group.position.z + nz*speed;
   856→      if (!checkEnemyWall(newX, e.group.position.z)) e.group.position.x = newX;
   857→      if (!checkEnemyWall(e.group.position.x, newZ)) e.group.position.z = newZ;
   858→      e.animTime += dt * 6;
   859→      e.leftArm.rotation.x = Math.sin(e.animTime) * 0.6;
   860→      e.rightArm.rotation.x = -Math.sin(e.animTime) * 0.6;
   861→      e.leftLeg.rotation.x = -Math.sin(e.animTime) * 0.5;
   862→      e.rightLeg.rotation.x = Math.sin(e.animTime) * 0.5;
   863→    } else {
   864→      e.attackTimer -= dt;
   865→      if (e.attackTimer <= 0) { e.attackTimer = ENEMY_ATK_CD; damagePlayer(ENEMY_DMG); }
   866→      e.animTime += dt * 8;
   867→      e.leftArm.rotation.x = -Math.abs(Math.sin(e.animTime)) * 1.3;
   868→      e.rightArm.rotation.x = -Math.abs(Math.sin(e.animTime)) * 1.3;
   869→      e.leftLeg.rotation.x = 0;
   870→      e.rightLeg.rotation.x = 0;
   871→    }
   872→  }
   873→}

  1219→// ===== COLLISION DETECTION =====
  1220→function checkCollision(px, py, pz) {
  1221→  const r = PLAYER_RADIUS;
  1222→  const feetY = py - PLAYER_HEIGHT;
  1223→  for (let i = 0; i < colliders.length; i++) {
  1224→    const c = colliders[i];
  1225→    if (px + r > c.minX && px - r < c.maxX &&
  1226→        pz + r > c.minZ && pz - r < c.maxZ) {
  1227→      // Only block if collider extends at least 0.3 above player feet
  1228→      // This prevents thin floor slabs from blocking movement
  1229→      if (c.maxY > feetY + 0.3 && c.minY < py) {
  1230→        return c;
  1231→      }
  1232→    }
  1233→  }
  1234→  return null;
  1235→}

  1237→// Find highest surface the player can stand on
  1238→function getGroundY(px, py, pz) {
  1239→  const r = PLAYER_RADIUS;
  1240→  const feetY = py - PLAYER_HEIGHT;
  1241→  let best = -999;
  1242→  const STEP_UP = 0.4;
  1243→  for (let i = 0; i < colliders.length; i++) {
  1244→    const c = colliders[i];
  1245→    if (px + r > c.minX && px - r < c.maxX &&
  1246→        pz + r > c.minZ && pz - r < c.maxZ) {
  1247→      // Only count surfaces near the TOP of the collider (player stands on it)
  1248→      // Not walls where feet are at the BOTTOM
  1249→      if (feetY >= c.maxY - STEP_UP && feetY <= c.maxY + 0.1 &&
  1250→          c.minY < py && c.maxY > best) {
  1251→        best = c.maxY;
  1252→      }
  1253→    }
  1254→  }
  1255→  return best;
  1256→}

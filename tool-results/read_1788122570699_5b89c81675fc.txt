   273→   273→  for (let i = 0; i < bufferSize; i++) {
   274→   274→    data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.15));
   275→   275→  }
   276→   276→  const src = audioCtx.createBufferSource();
   277→   277→  src.buffer = buffer;
   278→   278→  const filter = audioCtx.createBiquadFilter();

   281→   281→  const gain = audioCtx.createGain();
   282→   282→  gain.gain.setValueAtTime(vol, audioCtx.currentTime);

   284→   284→  src.connect(filter).connect(gain).connect(audioCtx.destination);

   288→   288→function playGunshot() {
   289→   289→  ensureAudio();
   290→   290→  const vol = masterVolume * 0.6;
   291→   291→  if (vol <= 0) return;
   292→   292→  const dur = 0.15;
   293→   293→  const bufferSize = audioCtx.sampleRate * dur;
   294→   294→  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
   295→   295→  const data = buffer.getChannelData(0);
   296→   296→  for (let i = 0; i < bufferSize; i++) {
   297→   297→    const t = i / audioCtx.sampleRate;
   298→   298→    data[i] = (Math.random() * 2 - 1) * Math.exp(-t * 30) * 1.5;
   299→   299→  }
   300→   300→  const src = audioCtx.createBufferSource();
   301→   301→  src.buffer = buffer;
   302→   302→  // Low thump
   303→   303→  const osc = audioCtx.createOscillator();
   304→   304→  osc.type = 'sine';
   305→   305→  osc.frequency.setValueAtTime(150, audioCtx.currentTime);
   306→   306→  osc.frequency.exponentialRampToValueAtTime(40, audioCtx.currentTime + 0.1);
   307→   307→  const oscGain = audioCtx.createGain();
   308→   308→  oscGain.gain.setValueAtTime(vol * 0.8, audioCtx.currentTime);
   309→   309→  oscGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
   310→   310→  // Noise part
   311→   311→  const noiseGain = audioCtx.createGain();
   312→   312→  noiseGain.gain.setValueAtTime(vol, audioCtx.currentTime);
   313→   313→  noiseGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
   314→   314→  const filter = audioCtx.createBiquadFilter();
   315→   315→  filter.type = 'bandpass';
   316→   316→  filter.frequency.value = 1000;
   317→   317→  filter.Q.value = 0.5;
   318→   318→  src.connect(filter).connect(noiseGain).connect(audioCtx.destination);
   319→   319→  osc.connect(oscGain).connect(audioCtx.destination);
   320→   320→  src.start();
   321→   321→  osc.start();
   322→   322→  osc.stop(audioCtx.currentTime + 0.1);
   323→   323→}

   781→   781→// ===== COLLISION DETECTION =====
   782→   782→function checkCollision(px, py, pz) {
   783→   783→  const r = PLAYER_RADIUS;
   784→   784→  const feetY = py - PLAYER_HEIGHT;
   785→   785→  for (let i = 0; i < colliders.length; i++) {
   786→   786→    const c = colliders[i];
   787→   787→    if (px + r > c.minX && px - r < c.maxX &&
   788→   788→        pz + r > c.minZ && pz - r < c.maxZ) {
   789→   789→      // Only block if collider extends at least 0.3 above player feet
   790→   790→      // This prevents thin floor slabs from blocking movement
   791→   791→      if (c.maxY > feetY + 0.3 && c.minY < py) {
   792→   792→        return c;
   793→   793→      }
   794→   794→    }
   795→   795→  }
   796→   796→  return null;
   797→   797→}

   799→   799→// Find highest surface the player can stand on
   800→   800→function getGroundY(px, py, pz) {
   801→   801→  const r = PLAYER_RADIUS;
   802→   802→  const feetY = py - PLAYER_HEIGHT;
   803→   803→  let best = -999;
   804→   804→  const STEP_UP = 0.4;
   805→   805→  for (let i = 0; i < colliders.length; i++) {
   806→   806→    const c = colliders[i];
   807→   807→    if (px + r > c.minX && px - r < c.maxX &&
   808→   808→        pz + r > c.minZ && pz - r < c.maxZ) {
   809→   809→      // Only count surfaces near the TOP of the collider (player stands on it)
   810→   810→      // Not walls where feet are at the BOTTOM
   811→   811→      if (feetY >= c.maxY - STEP_UP && feetY <= c.maxY + 0.1 &&
   812→   812→          c.minY < py && c.maxY > best) {
   813→   813→        best = c.maxY;
   814→   814→      }
   815→   815→    }
   816→   816→  }
   817→   817→  return best;
   818→   818→}

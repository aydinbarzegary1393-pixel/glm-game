   324→   324→function playFootstep() {
   325→   325→  ensureAudio();
   326→   326→  const vol = masterVolume * 0.3;
   327→   327→  if (vol <= 0) return;
   328→   328→  // Create short noise burst for footstep
   329→   329→  const bufferSize = audioCtx.sampleRate * 0.08;
   330→   330→  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
   331→   331→  const data = buffer.getChannelData(0);
   332→   332→  for (let i = 0; i < bufferSize; i++) {
   333→   333→    data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.15));
   334→   334→  }
   335→   335→  const src = audioCtx.createBufferSource();
   336→   336→  src.buffer = buffer;
   337→   337→  const filter = audioCtx.createBiquadFilter();
   338→   338→  filter.type = 'lowpass';
   339→   339→  filter.frequency.value = 800;
   340→   340→  const gain = audioCtx.createGain();
   341→   341→  gain.gain.setValueAtTime(vol, audioCtx.currentTime);
   342→   342→  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.08);
   343→   343→  src.connect(filter).connect(gain).connect(audioCtx.destination);
   344→   344→  src.start();
   345→   345→}

   347→   347→function playGunshot() {
   348→   348→  ensureAudio();
   349→   349→  const vol = masterVolume * 0.6;
   350→   350→  if (vol <= 0) return;
   351→   351→  const dur = 0.15;
   352→   352→  const bufferSize = audioCtx.sampleRate * dur;
   353→   353→  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
   354→   354→  const data = buffer.getChannelData(0);
   355→   355→  for (let i = 0; i < bufferSize; i++) {
   356→   356→    const t = i / audioCtx.sampleRate;
   357→   357→    data[i] = (Math.random() * 2 - 1) * Math.exp(-t * 30) * 1.5;
   358→   358→  }
   359→   359→  const src = audioCtx.createBufferSource();
   360→   360→  src.buffer = buffer;
   361→   361→  // Low thump
   362→   362→  const osc = audioCtx.createOscillator();
   363→   363→  osc.type = 'sine';
   364→   364→  osc.frequency.setValueAtTime(150, audioCtx.currentTime);
   365→   365→  osc.frequency.exponentialRampToValueAtTime(40, audioCtx.currentTime + 0.1);
   366→   366→  const oscGain = audioCtx.createGain();
   367→   367→  oscGain.gain.setValueAtTime(vol * 0.8, audioCtx.currentTime);
   368→   368→  oscGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
   369→   369→  // Noise part
   370→   370→  const noiseGain = audioCtx.createGain();
   371→   371→  noiseGain.gain.setValueAtTime(vol, audioCtx.currentTime);
   372→   372→  noiseGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
   373→   373→  const filter = audioCtx.createBiquadFilter();
   374→   374→  filter.type = 'bandpass';
   375→   375→  filter.frequency.value = 1000;
   376→   376→  filter.Q.value = 0.5;
   377→   377→  src.connect(filter).connect(noiseGain).connect(audioCtx.destination);
   378→   378→  osc.connect(oscGain).connect(audioCtx.destination);
   379→   379→  src.start();
   380→   380→  osc.start();
   381→   381→  osc.stop(audioCtx.currentTime + 0.1);
   382→   382→}

   724→   724→  var px = player.position.x, pz = player.position.z;

   726→   726→    var angle, dist, ex, ez, attempts = 0;
   727→   727→    do {
   728→   728→      angle = (i / count) * Math.PI * 2 + Math.random() * 0.5;
   729→   729→      dist = 8 + Math.random() * 10;
   730→   730→      ex = px + Math.cos(angle) * dist;
   731→   731→      ez = pz + Math.sin(angle) * dist;
   732→   732→      ex = Math.max(-15, Math.min(15, ex));
   733→   733→      ez = Math.max(-15, Math.min(15, ez));
   734→   734→      attempts++;
   735→   735→    } while (checkEnemyWall(ex, ez) && attempts < 10);
   736→   736→    createEnemy(ex, ez);
   737→   737→  }
   738→   738→}

   740→   740→function updateEnemies(dt) {
   741→   741→  var pPos = player.position;
   742→   742→  for (var i = enemies.length - 1; i >= 0; i--) {
   743→   743→    var e = enemies[i];
   744→   744→    var dx = pPos.x - e.group.position.x;
   745→   745→    var dz = pPos.z - e.group.position.z;
   746→   746→    var dist = Math.sqrt(dx*dx + dz*dz);
   747→   747→    if (dist < 0.1) dist = 0.1;
   748→   748→    e.group.rotation.y = Math.atan2(dx, dz);
   749→   749→    if (dist > ENEMY_ATK_RANGE) {
   750→   750→      var speed = ENEMY_SPEED * dt;
   751→   751→      var nx = dx/dist, nz = dz/dist;
   752→   752→      var newX = e.group.position.x + nx*speed;
   753→   753→      var newZ = e.group.position.z + nz*speed;
   754→   754→      if (!checkEnemyWall(newX, e.group.position.z)) e.group.position.x = newX;
   755→   755→      if (!checkEnemyWall(e.group.position.x, newZ)) e.group.position.z = newZ;
   756→   756→      e.animTime += dt * 6;
   757→   757→      e.leftArm.rotation.x = Math.sin(e.animTime) * 0.6;
   758→   758→      e.rightArm.rotation.x = -Math.sin(e.animTime) * 0.6;
   759→   759→      e.leftLeg.rotation.x = -Math.sin(e.animTime) * 0.5;
   760→   760→      e.rightLeg.rotation.x = Math.sin(e.animTime) * 0.5;
   761→   761→    } else {
   762→   762→      e.attackTimer -= dt;
   763→   763→      if (e.attackTimer <= 0) { e.attackTimer = ENEMY_ATK_CD; damagePlayer(ENEMY_DMG); }
   764→   764→      e.animTime += dt * 8;
   765→   765→      e.leftArm.rotation.x = -Math.abs(Math.sin(e.animTime)) * 1.3;
   766→   766→      e.rightArm.rotation.x = -Math.abs(Math.sin(e.animTime)) * 1.3;
   767→   767→      e.leftLeg.rotation.x = 0;
   768→   768→      e.rightLeg.rotation.x = 0;
   769→   769→    }
   770→   770→  }
   771→   771→}

  1094→  1094→function checkCollision(px, py, pz) {
  1095→  1095→  const r = PLAYER_RADIUS;
  1096→  1096→  const feetY = py - PLAYER_HEIGHT;
  1097→  1097→  for (let i = 0; i < colliders.length; i++) {
  1098→  1098→    const c = colliders[i];
  1099→  1099→    if (px + r > c.minX && px - r < c.maxX &&
  1100→  1100→        pz + r > c.minZ && pz - r < c.maxZ) {
  1101→  1101→      // Only block if collider extends at least 0.3 above player feet
  1102→  1102→      // This prevents thin floor slabs from blocking movement
  1103→  1103→      if (c.maxY > feetY + 0.3 && c.minY < py) {
  1104→  1104→        return c;
  1105→  1105→      }
  1106→  1106→    }
  1107→  1107→  }
  1108→  1108→  return null;
  1109→  1109→}

  1111→  1111→// Find highest surface the player can stand on
  1112→  1112→function getGroundY(px, py, pz) {
  1113→  1113→  const r = PLAYER_RADIUS;
  1114→  1114→  const feetY = py - PLAYER_HEIGHT;
  1115→  1115→  let best = -999;
  1116→  1116→  const STEP_UP = 0.4;
  1117→  1117→  for (let i = 0; i < colliders.length; i++) {
  1118→  1118→    const c = colliders[i];
  1119→  1119→    if (px + r > c.minX && px - r < c.maxX &&
  1120→  1120→        pz + r > c.minZ && pz - r < c.maxZ) {
  1121→  1121→      // Only count surfaces near the TOP of the collider (player stands on it)
  1122→  1122→      // Not walls where feet are at the BOTTOM
  1123→  1123→      if (feetY >= c.maxY - STEP_UP && feetY <= c.maxY + 0.1 &&
  1124→  1124→          c.minY < py && c.maxY > best) {
  1125→  1125→        best = c.maxY;
  1126→  1126→      }
  1127→  1127→    }
  1128→  1128→  }
  1129→  1129→  return best;
  1130→  1130→}

   252→// ===== AUDIO SYSTEM (Web Audio API) =====
   253→let audioCtx = null;
   254→let gunAudioCtx = null;
   255→
   256→function ensureAudio() {
   257→  if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
   258→}
   259→
   260→function playFootstep() {
   261→  ensureAudio();
   262→  const vol = masterVolume * 0.3;
   263→  if (vol <= 0) return;
   264→  // Create short noise burst for footstep
   265→  const bufferSize = audioCtx.sampleRate * 0.08;
   266→  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
   267→  const data = buffer.getChannelData(0);
   268→  for (let i = 0; i < bufferSize; i++) {
   269→    data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.15));
   270→  }
   271→  const src = audioCtx.createBufferSource();
   272→  src.buffer = buffer;
   273→  const filter = audioCtx.createBiquadFilter();
   274→  filter.type = 'lowpass';
   275→  filter.frequency.value = 800;
   276→  const gain = audioCtx.createGain();
   277→  gain.gain.setValueAtTime(vol, audioCtx.currentTime);
   278→  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.08);
   279→  src.connect(filter).connect(gain).connect(audioCtx.destination);
   280→  src.start();
   281→}
   282→
   283→function playGunshot() {
   284→  ensureAudio();
   285→  const vol = masterVolume * 0.6;
   286→  if (vol <= 0) return;
   287→  const dur = 0.15;
   288→  const bufferSize = audioCtx.sampleRate * dur;
   289→  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
   290→  const data = buffer.getChannelData(0);
   291→  for (let i = 0; i < bufferSize; i++) {
   292→    const t = i / audioCtx.sampleRate;
   293→    data[i] = (Math.random() * 2 - 1) * Math.exp(-t * 30) * 1.5;
   294→  }
   295→  const src = audioCtx.createBufferSource();
   296→  src.buffer = buffer;
   297→  // Low thump
   298→  const osc = audioCtx.createOscillator();
   299→  osc.type = 'sine';
   300→  osc.frequency.setValueAtTime(150, audioCtx.currentTime);
   301→  osc.frequency.exponentialRampToValueAtTime(40, audioCtx.currentTime + 0.1);
   302→  const oscGain = audioCtx.createGain();
   303→  oscGain.gain.setValueAtTime(vol * 0.8, audioCtx.currentTime);
   304→  oscGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
   305→  // Noise part
   306→  const noiseGain = audioCtx.createGain();
   307→  noiseGain.gain.setValueAtTime(vol, audioCtx.currentTime);
   308→  noiseGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
   309→  const filter = audioCtx.createBiquadFilter();
   310→  filter.type = 'bandpass';
   311→  filter.frequency.value = 1000;
   312→  filter.Q.value = 0.5;
   313→  src.connect(filter).connect(noiseGain).connect(audioCtx.destination);
   314→  osc.connect(oscGain).connect(audioCtx.destination);
   315→  src.start();
   316→  osc.start();
   317→  osc.stop(audioCtx.currentTime + 0.1);
   318→}

   320→function playReloadSound() {
   321→  ensureAudio();
   322→  const vol = masterVolume * 0.4;
   323→  if (vol <= 0) return;
   324→  // Click sound
   325→  var buf = audioCtx.createBuffer(1, audioCtx.sampleRate * 0.05, audioCtx.sampleRate);
   326→  var d = buf.getChannelData(0);
   327→  for (var i = 0; i < d.length; i++) d[i] = (Math.random() * 2 - 1) * Math.exp(-i / (d.length * 0.1));

   376→function initJoystick() {

   380→  zone.addEventListener('touchstart', function(e) {
   381→    e.preventDefault();

   383→    joyTouchId = t.identifier;
   384→    var rect = joyBase.getBoundingClientRect();
   385→    joyBaseX = rect.left + rect.width / 2;
   386→    joyBaseY = rect.top + rect.height / 2;
   387→  }, {passive: false});
   388→  zone.addEventListener('touchmove', function(e) {
   389→    e.preventDefault();
   390→    for (var i = 0; i < e.changedTouches.length; i++) {
   391→      var t = e.changedTouches[i];
   392→      if (t.identifier === joyTouchId) {
   393→        var dx = t.clientX - joyBaseX;
   394→        var dy = t.clientY - joyBaseY;
   395→        var dist = Math.sqrt(dx * dx + dy * dy);
   396→        if (dist > JOY_RADIUS) { dx = dx / dist * JOY_RADIUS; dy = dy / dist * JOY_RADIUS; }
   397→        joyInputX = dx / JOY_RADIUS;
   398→        joyInputY = dy / JOY_RADIUS;
   399→        joyStick.style.transform = 'translate(' + dx + 'px,' + dy + 'px)';
   400→      }
   401→    }
   402→  }, {passive: false});
   403→  var resetJoy = function(e) {
   404→    for (var i = 0; i < e.changedTouches.length; i++) {
   405→      if (e.changedTouches[i].identifier === joyTouchId) {
   406→        joyTouchId = null; joyInputX = 0; joyInputY = 0;
   407→        joyStick.style.transform = 'translate(0,0)';
   408→      }
   409→    }
   410→  };
   411→  zone.addEventListener('touchend', resetJoy);
   412→  zone.addEventListener('touchcancel', resetJoy);

   759→// ===== COLLISION DETECTION =====
   760→function checkCollision(px, py, pz) {
   761→  const r = PLAYER_RADIUS;
   762→  const feetY = py - PLAYER_HEIGHT;
   763→  for (let i = 0; i < colliders.length; i++) {
   764→    const c = colliders[i];
   765→    if (px + r > c.minX && px - r < c.maxX &&
   766→        pz + r > c.minZ && pz - r < c.maxZ) {
   767→      // Only block if collider extends at least 0.3 above player feet
   768→      // This prevents thin floor slabs from blocking movement
   769→      if (c.maxY > feetY + 0.3 && c.minY < py) {
   770→        return c;
   771→      }
   772→    }
   773→  }
   774→  return null;
   775→}

   777→// Find highest surface the player can stand on
   778→function getGroundY(px, py, pz) {
   779→  const r = PLAYER_RADIUS;
   780→  const feetY = py - PLAYER_HEIGHT;
   781→  let best = -999;
   782→  const STEP_UP = 0.4;
   783→  for (let i = 0; i < colliders.length; i++) {
   784→    const c = colliders[i];
   785→    if (px + r > c.minX && px - r < c.maxX &&
   786→        pz + r > c.minZ && pz - r < c.maxZ) {
   787→      // Only count surfaces near the TOP of the collider (player stands on it)
   788→      // Not walls where feet are at the BOTTOM
   789→      if (feetY >= c.maxY - STEP_UP && feetY <= c.maxY + 0.1 &&
   790→          c.minY < py && c.maxY > best) {
   791→        best = c.maxY;
   792→      }
   793→    }
   794→  }
   795→  return best;
   796→}

  1386→function animate() {
  1387→  requestAnimationFrame(animate);
  1388→  const dt = Math.min(clock.getDelta(), 0.05);
  1389→
  1390→  if (isLocked || (isMobileDevice && gameStarted)) {
  1391→    // Movement
  1392→    const fwd = new THREE.Vector3(0,0,-1).applyAxisAngle(new THREE.Vector3(0,1,0), yaw);
  1393→    const rgt = new THREE.Vector3(1,0,0).applyAxisAngle(new THREE.Vector3(0,1,0), yaw);
  1394→    const mv = new THREE.Vector3();
  1395→    if(keys.w) mv.add(fwd); if(keys.s) mv.sub(fwd);
  1396→    if(keys.d) mv.add(rgt); if(keys.a) mv.sub(rgt);
  1397→    // Mobile joystick input
  1398→    if (isMobileDevice && (Math.abs(joyInputX) > 0.1 || Math.abs(joyInputY) > 0.1)) {
  1399→      mv.add(fwd.clone().multiplyScalar(-joyInputY));
  1400→      mv.add(rgt.clone().multiplyScalar(joyInputX));
  1401→    }
  1402→    if(mv.lengthSq()>0) mv.normalize();
  1403→
  1404→    const speed = MOVE_SPEED * dt;
  1405→    const px = player.position.x;
  1406→    const py = player.position.y;
  1407→    const pz = player.position.z;
  1408→
  1409→    // Try X movement (with collision)
  1410→    let newX = px + mv.x * speed;
  1411→    if (!checkCollision(newX, py, pz)) {
  1412→      player.position.x = newX;
  1413→    }
  1414→
  1415→    // Try Z movement (with collision)
  1416→    let newZ = pz + mv.z * speed;
  1417→    if (!checkCollision(player.position.x, py, newZ)) {
  1418→      player.position.z = newZ;
  1419→    }
  1420→
  1421→    const moving = mv.lengthSq() > 0;
  1422→
  1423→    // Jump (only if standing on ground or a surface)
  1424→    var rawGround = getGroundY(player.position.x, player.position.y, player.position.z);
  1425→    var groundLevel = rawGround > -998 ? rawGround : 0;
  1426→    var onGround = (player.position.y - PLAYER_HEIGHT <= groundLevel + 0.05 && velocityY <= 0.1);
  1427→    if (keys.space && onGround) {
  1428→      velocityY = JUMP_FORCE;
  1429→    }
  1430→    // Gravity
  1431→    velocityY -= GRAVITY * dt;
  1432→    player.position.y += velocityY * dt;
  1433→    // Ground collision: stand on boxes/floors
  1434→    rawGround = getGroundY(player.position.x, player.position.y, player.position.z);
  1435→    groundLevel = rawGround > -998 ? rawGround : 0;
  1436→    if (player.position.y - PLAYER_HEIGHT <= groundLevel + 0.05) {
  1437→      player.position.y = groundLevel + PLAYER_HEIGHT;
  1438→      velocityY = 0;
  1439→    }
  1440→
  1441→    // Footstep sounds
  1442→    if (moving) {
  1443→      footstepTimer -= dt;
  1444→      if (footstepTimer <= 0) {
  1445→        playFootstep();
  1446→        footstepTimer = FOOTSTEP_INTERVAL;
  1447→      }
  1448→    } else {
  1449→      footstepTimer = 0;
  1450→    }
  1451→
  1452→    // Gun recoil
  1453→    if (recoilAmount > 0) {
  1454→      recoilAmount *= 0.85;
  1455→      if (recoilAmount < 0.001) recoilAmount = 0;
  1456→    }
  1457→
  1458→    // Gun bob when walking
  1459→    let bobY = 0, bobX = 0;
  1460→    if (moving) {
  1461→      const bobSpeed = 10;
  1462→      const bobAmount = 0.008;
  1463→      bobY = Math.sin(clock.elapsedTime * bobSpeed) * bobAmount;
  1464→      bobX = Math.cos(clock.elapsedTime * bobSpeed * 0.5) * bobAmount * 0.5;
  1465→    }
  1466→    gunGroup.position.y = -0.22 + bobY + recoilAmount;
  1467→    gunGroup.position.x = 0.25 + bobX;
  1468→
  1469→    // Debug + ammo HUD
  1470→    debugEl.innerHTML =
  1471→      'Pos: '+player.position.x.toFixed(1)+', '+player.position.y.toFixed(1)+', '+player.position.z.toFixed(1)+
  1472→      '<br>Yaw: '+(yaw*180/Math.PI).toFixed(1)+'°  Pitch: '+(pitch*180/Math.PI).toFixed(1)+'°'+
  1473→      '<br>Ammo: ' + ammo + ' / ' + maxAmmo + (isReloading ? ' <span style="color:#ffaa00">[RELOADING]</span>' : '');
  1474→  }

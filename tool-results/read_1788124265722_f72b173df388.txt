   264→// ===== AUDIO SYSTEM (Web Audio API) =====
   265→let audioCtx = null;
   266→let gunAudioCtx = null;
   267→
   268→function ensureAudio() {
   269→  if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
   270→}
   271→
   272→function playFootstep() {
   273→  ensureAudio();
   274→  const vol = masterVolume * 0.3;
   275→  if (vol <= 0) return;
   276→  // Create short noise burst for footstep
   277→  const bufferSize = audioCtx.sampleRate * 0.08;
   278→  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
   279→  const data = buffer.getChannelData(0);
   280→  for (let i = 0; i < bufferSize; i++) {
   281→    data[i] = (Math.random() * 2 - 1) * Math.exp(-i / (bufferSize * 0.15));
   282→  }
   283→  const src = audioCtx.createBufferSource();
   284→  src.buffer = buffer;
   285→  const filter = audioCtx.createBiquadFilter();
   286→  filter.type = 'lowpass';
   287→  filter.frequency.value = 800;
   288→  const gain = audioCtx.createGain();
   289→  gain.gain.setValueAtTime(vol, audioCtx.currentTime);
   290→  gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.08);
   291→  src.connect(filter).connect(gain).connect(audioCtx.destination);
   292→  src.start();
   293→}
   294→
   295→function playGunshot() {
   296→  ensureAudio();
   297→  const vol = masterVolume * 0.6;
   298→  if (vol <= 0) return;
   299→  const dur = 0.15;
   300→  const bufferSize = audioCtx.sampleRate * dur;
   301→  const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
   302→  const data = buffer.getChannelData(0);
   303→  for (let i = 0; i < bufferSize; i++) {
   304→    const t = i / audioCtx.sampleRate;
   305→    data[i] = (Math.random() * 2 - 1) * Math.exp(-t * 30) * 1.5;
   306→  }
   307→  const src = audioCtx.createBufferSource();
   308→  src.buffer = buffer;
   309→  // Low thump
   310→  const osc = audioCtx.createOscillator();
   311→  osc.type = 'sine';
   312→  osc.frequency.setValueAtTime(150, audioCtx.currentTime);
   313→  osc.frequency.exponentialRampToValueAtTime(40, audioCtx.currentTime + 0.1);
   314→  const oscGain = audioCtx.createGain();
   315→  oscGain.gain.setValueAtTime(vol * 0.8, audioCtx.currentTime);
   316→  oscGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.1);
   317→  // Noise part
   318→  const noiseGain = audioCtx.createGain();
   319→  noiseGain.gain.setValueAtTime(vol, audioCtx.currentTime);
   320→  noiseGain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + dur);
   321→  const filter = audioCtx.createBiquadFilter();
   322→  filter.type = 'bandpass';
   323→  filter.frequency.value = 1000;
   324→  filter.Q.value = 0.5;
   325→  src.connect(filter).connect(noiseGain).connect(audioCtx.destination);
   326→  osc.connect(oscGain).connect(audioCtx.destination);
   327→  src.start();
   328→  osc.start();
   329→  osc.stop(audioCtx.currentTime + 0.1);
   330→}

   332→function playReloadSound() {
   333→  ensureAudio();
   334→  const vol = masterVolume * 0.4;
   335→  if (vol <= 0) return;
   336→  // Click sound
   337→  var buf = audioCtx.createBuffer(1, audioCtx.sampleRate * 0.05, audioCtx.sampleRate);
   338→  var d = buf.getChannelData(0);
   339→  for (var i = 0; i < d.length; i++) d[i] = (Math.random() * 2 - 1) * Math.exp(-i / (d.length * 0.1));
   340→  var s = audioCtx.createBufferSource(); s.buffer = buf;
   341→  var g = audioCtx.createGain(); g.gain.setValueAtTime(vol, audioCtx.currentTime);
   342→  var f = audioCtx.createBiquadFilter(); f.type = 'bandpass'; f.frequency.value = 2000; f.Q.value = 2;
   343→  s.connect(f).connect(g).connect(audioCtx.destination); s.start();
   344→  // Magazine slide sound after delay
   345→  setTimeout(function() {
   346→    var buf2 = audioCtx.createBuffer(1, audioCtx.sampleRate * 0.15, audioCtx.sampleRate);
   347→    var d2 = buf2.getChannelData(0);
   348→    for (var i = 0; i < d2.length; i++) d2[i] = (Math.random() * 2 - 1) * Math.exp(-i / (d2.length * 0.15)) * 0.6;
   349→    var s2 = audioCtx.createBufferSource(); s2.buffer = buf2;
   350→    var g2 = audioCtx.createGain(); g2.gain.setValueAtTime(vol * 0.7, audioCtx.currentTime);
   351→    var f2 = audioCtx.createBiquadFilter(); f2.type = 'bandpass'; f2.frequency.value = 1500; f2.Q.value = 1;
   352→    s2.connect(f2).connect(g2).connect(audioCtx.destination); s2.start();
   353→  }, 300);
   354→}

   822→// ===== COLLISION DETECTION =====
   823→function checkCollision(px, py, pz) {
   824→  const r = PLAYER_RADIUS;
   825→  const feetY = py - PLAYER_HEIGHT;
   826→  for (let i = 0; i < colliders.length; i++) {
   827→    const c = colliders[i];
   828→    if (px + r > c.minX && px - r < c.maxX &&
   829→        pz + r > c.minZ && pz - r < c.maxZ) {
   830→      // Only block if collider extends at least 0.3 above player feet
   831→      // This prevents thin floor slabs from blocking movement
   832→      if (c.maxY > feetY + 0.3 && c.minY < py) {
   833→        return c;
   834→      }
   835→    }
   836→  }
   837→  return null;
   838→}

   840→// Find highest surface the player can stand on
   841→function getGroundY(px, py, pz) {
   842→  const r = PLAYER_RADIUS;
   843→  const feetY = py - PLAYER_HEIGHT;
   844→  let best = -999;
   845→  const STEP_UP = 0.4;
   846→  for (let i = 0; i < colliders.length; i++) {
   847→    const c = colliders[i];
   848→    if (px + r > c.minX && px - r < c.maxX &&
   849→        pz + r > c.minZ && pz - r < c.maxZ) {
   850→      // Only count surfaces near the TOP of the collider (player stands on it)
   851→      // Not walls where feet are at the BOTTOM
   852→      if (feetY >= c.maxY - STEP_UP && feetY <= c.maxY + 0.1 &&
   853→          c.minY < py && c.maxY > best) {
   854→        best = c.maxY;
   855→      }
   856→    }
   857→  }
   858→  return best;
   859→}

  1299→function shoot() {
  1300→  if (!canShoot || isReloading || ammo <= 0) return;
  1301→  canShoot = false;
  1302→  ammo--;
  1303→
  1304→  // Sound
  1305→  playGunshot();
  1306→
  1307→  // Muzzle flash
  1308→  muzzleFlashEl.classList.remove('fire');
  1309→  void muzzleFlashEl.offsetWidth;
  1310→  muzzleFlashEl.classList.add('fire');
  1311→  muzzleLight.intensity = 3;
  1312→  setTimeout(() => { muzzleLight.intensity = 0; }, 50);
  1313→
  1314→  // Gun recoil (Three.js, not DOM)
  1315→  recoilAmount = 0.06;
  1316→
  1317→  // Raycast
  1318→  raycaster.setFromCamera(new THREE.Vector2(0, 0), camera);
  1319→  const hits = raycaster.intersectObjects(sceneObjects);
  1320→  if (hits.length > 0) {
  1321→    const hit = hits[0];
  1322→    // Convert face normal to world space (fixes rotated objects like ground plane)
  1323→    const n = hit.face.normal.clone().transformDirection(hit.object.matrixWorld).normalize();
  1324→    // Bullet hole: dark center + lighter ring
  1325→    const holeGroup = new THREE.Group();
  1326→    const coreGeo = new THREE.CircleGeometry(0.06, 8);
  1327→    const coreMat = new THREE.MeshBasicMaterial({color:0x111111, side:THREE.DoubleSide, depthWrite:false});
  1328→    const core = new THREE.Mesh(coreGeo, coreMat);
  1329→    holeGroup.add(core);
  1330→    const ringGeo = new THREE.RingGeometry(0.06, 0.1, 12);
  1331→    const ringMat = new THREE.MeshBasicMaterial({color:0x333333, side:THREE.DoubleSide, depthWrite:false, transparent:true, opacity:0.6});
  1332→    const ring = new THREE.Mesh(ringGeo, ringMat);
  1333→    holeGroup.add(ring);
  1334→    holeGroup.position.copy(hit.point);
  1335→    holeGroup.lookAt(hit.point.clone().add(n));
  1336→    holeGroup.position.add(n.clone().multiplyScalar(0.02));
  1337→    holeGroup.renderOrder = 999;
  1338→    bulletHoles.add(holeGroup);
  1339→  }
  1340→
  1341→  // Fire rate cooldown
  1342→  setTimeout(() => { canShoot = true; }, 120);
  1343→}

  1475→function animate() {

  1477→  const dt = Math.min(clock.getDelta(), 0.05);
  1478→
  1479→  if (isLocked || (isMobileDevice && gameStarted)) {
  1480→    // Movement
  1481→    const fwd = new THREE.Vector3(0,0,-1).applyAxisAngle(new THREE.Vector3(0,1,0), yaw);
  1482→    const rgt = new THREE.Vector3(1,0,0).applyAxisAngle(new THREE.Vector3(0,1,0), yaw);
  1483→    const mv = new THREE.Vector3();

  1485→    if(keys.d) mv.add(rgt); if(keys.a) mv.sub(rgt);

  1487→    if (isMobileDevice && (Math.abs(joyInputX) > 0.1 || Math.abs(joyInputY) > 0.1)) {
  1488→      mv.add(fwd.clone().multiplyScalar(-joyInputY));
  1489→      mv.add(rgt.clone().multiplyScalar(joyInputX));
  1490→    }
  1491→    if(mv.lengthSq()>0) mv.normalize();
  1492→
  1493→    const speed = MOVE_SPEED * dt;
  1494→    const px = player.position.x;
  1495→    const py = player.position.y;
  1496→    const pz = player.position.z;
  1497→
  1498→    // Try X movement (with collision)
  1499→    let newX = px + mv.x * speed;
  1500→    if (!checkCollision(newX, py, pz)) {
  1501→      player.position.x = newX;
  1502→    }
  1503→
  1504→    // Try Z movement (with collision)
  1505→    let newZ = pz + mv.z * speed;
  1506→    if (!checkCollision(player.position.x, py, newZ)) {
  1507→      player.position.z = newZ;
  1508→    }
  1509→
  1510→    const moving = mv.lengthSq() > 0;
  1511→
  1512→    // Jump (only if standing on ground or a surface)
  1513→    var rawGround = getGroundY(player.position.x, player.position.y, player.position.z);
  1514→    var groundLevel = rawGround > -998 ? rawGround : 0;
  1515→    var onGround = (player.position.y - PLAYER_HEIGHT <= groundLevel + 0.05 && velocityY <= 0.1);
  1516→    if (keys.space && onGround) {
  1517→      velocityY = JUMP_FORCE;
  1518→    }
  1519→    // Gravity
  1520→    velocityY -= GRAVITY * dt;
  1521→    player.position.y += velocityY * dt;
  1522→    // Ground collision: stand on boxes/floors
  1523→    rawGround = getGroundY(player.position.x, player.position.y, player.position.z);
  1524→    groundLevel = rawGround > -998 ? rawGround : 0;
  1525→    if (player.position.y - PLAYER_HEIGHT <= groundLevel + 0.05) {
  1526→      player.position.y = groundLevel + PLAYER_HEIGHT;
  1527→      velocityY = 0;
  1528→    }
  1529→
  1530→    // Footstep sounds
  1531→    if (moving) {
  1532→      footstepTimer -= dt;
  1533→      if (footstepTimer <= 0) {
  1534→        playFootstep();
  1535→        footstepTimer = FOOTSTEP_INTERVAL;
  1536→      }
  1537→    } else {
  1538→      footstepTimer = 0;
  1539→    }
  1540→
  1541→    // Gun recoil
  1542→    if (recoilAmount > 0) {
  1543→      recoilAmount *= 0.85;
  1544→      if (recoilAmount < 0.001) recoilAmount = 0;
  1545→    }
  1546→
  1547→    // Gun bob when walking
  1548→    let bobY = 0, bobX = 0;
  1549→    if (moving) {
  1550→      const bobSpeed = 10;

  1553→      bobX = Math.cos(clock.elapsedTime * bobSpeed * 0.5) * bobAmount * 0.5;
  1554→    }
  1555→    gunGroup.position.y = -0.22 + bobY + recoilAmount;
  1556→    gunGroup.position.x = 0.25 + bobX;
  1557→
  1558→    // Ammo HUD
  1559→    if (isReloading) {

  1563→    }
  1564→  }

  1566→  renderer.render(scene, camera);
  1567→}
